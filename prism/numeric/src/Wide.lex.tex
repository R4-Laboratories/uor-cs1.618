definition{
  name: round,
  parameters: [w : Limbs, amount : Int16],
  result: (Option UInt64),
  body: (if (negative amount) then (match (int16Negate amount)
    | Option.some positive => (match (convertInt16ToUInt32 positive)
      | Option.some count => (match (low64 w)
        | Option.some bits => (prim shift_left bits count : (Option UInt64))
        | Option.none => (ctor Option.none [UInt64]))
      | Option.none => (ctor Option.none [UInt64]))
    | Option.none => (ctor Option.none [UInt64])) else (if (signedLess (128 : Int16) amount) then (ctor Option.some [UInt64] (0 : UInt64)) else (match (convertInt16ToUInt8 amount)
    | Option.some count => (match (match (ctor Option.some [Rounding] (record Rounding {value := w, guard := false, sticky := false}))
        | Option.some state0 => (match (if (prim equal (prim bit_and count (1 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state0) else (roundStep (ctor Option.some [Rounding] state0)))
          | Option.some state1 => (match (if (prim equal (prim bit_and count (2 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state1) else (round2 (ctor Option.some [Rounding] state1)))
            | Option.some state2 => (match (if (prim equal (prim bit_and count (4 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state2) else (round4 (ctor Option.some [Rounding] state2)))
              | Option.some state3 => (match (if (prim equal (prim bit_and count (8 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state3) else (round8 (ctor Option.some [Rounding] state3)))
                | Option.some state4 => (match (if (prim equal (prim bit_and count (16 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state4) else (round16 (ctor Option.some [Rounding] state4)))
                  | Option.some state5 => (match (if (prim equal (prim bit_and count (32 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state5) else (round32 (ctor Option.some [Rounding] state5)))
                    | Option.some state6 => (match (if (prim equal (prim bit_and count (64 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state6) else (round64 (ctor Option.some [Rounding] state6)))
                      | Option.some state7 => (match (if (prim equal (prim bit_and count (128 : UInt8) : UInt8) (0 : UInt8) : Bool) then (ctor Option.some [Rounding] state7) else (round128 (ctor Option.some [Rounding] state7)))
                        | Option.some state8 => (ctor Option.some [Rounding] state8)
                        | Option.none => (ctor Option.none [Rounding]))
                      | Option.none => (ctor Option.none [Rounding]))
                    | Option.none => (ctor Option.none [Rounding]))
                  | Option.none => (ctor Option.none [Rounding]))
                | Option.none => (ctor Option.none [Rounding]))
              | Option.none => (ctor Option.none [Rounding]))
            | Option.none => (ctor Option.none [Rounding]))
          | Option.none => (ctor Option.none [Rounding]))
        | Option.none => (ctor Option.none [Rounding]))
      | Option.some st => (match (low64 (project st value))
        | Option.some bits => (if ((project st guard) && ((project st sticky) || (prim equal (prim bit_and bits (1 : UInt64) : UInt64) (1 : UInt64) : Bool))) then (prim checked_add bits (1 : UInt64) : (Option UInt64)) else (ctor Option.some [UInt64] bits))
        | Option.none => (ctor Option.none [UInt64]))
      | Option.none => (ctor Option.none [UInt64]))
    | Option.none => (ctor Option.none [UInt64])))),
  axioms: [Quot.sound, propext],
  executable: true
}

      | Option.some st => (ctor Option.some [UInt64] (prim bit_or (project st quotient) (if (prim equal (project st remainder) (0 : UInt64) : Bool) then (0 : UInt64) else (1 : UInt64)) : UInt64))
